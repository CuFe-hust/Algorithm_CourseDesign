# P37 数独求解

- **标签**：#搜索与回溯 #数独 #剪枝
- **难度**：★★★★★
- **来源**：经典算法题（Sudoku Solver）

## 题目描述

给定一个 9 x 9 的数独棋盘，其中数字 `0` 表示空格。请把空格填上 1 到 9 的数字，使得：

- 每一行恰好包含 1 到 9 各一次；
- 每一列恰好包含 1 到 9 各一次；
- 每个 3 x 3 的宫恰好包含 1 到 9 各一次。

这里的宫由棋盘按行、列各划分为三段得到：左上角宫的格子为第 1 到 3 行与第 1 到 3 列的交集，其余各宫同理。输入保证有且仅有一组合法填法。

## 输入格式

共 9 行，每行一个长度为 9 的字符串，只包含数字字符 `0` 到 `9`。`0` 表示空格，字符之间没有空格。

## 输出格式

共 9 行，每行一个长度为 9 的字符串，表示填好的完整数独。字符之间没有空格，每行末尾需要换行。

## 标准实现

### C

```c
#include <stdio.h>

char board[9][10];
int rowMask[9], colMask[9], boxMask[9];
const int FULL = (1 << 9) - 1;

int count_bits(int x) {
    int cnt = 0;
    while (x) {
        cnt += x & 1;
        x >>= 1;
    }
    return cnt;
}

// 位掩码记录已用数字，并优先选择候选最少的空格
int dfs(void) {
    int bestR = -1, bestC = -1, bestMask = 0, bestCnt = 10;
    for (int r = 0; r < 9; r++) {
        for (int c = 0; c < 9; c++) {
            if (board[r][c] != '0') continue;
            int b = (r / 3) * 3 + c / 3;
            int m = FULL & ~(rowMask[r] | colMask[c] | boxMask[b]);
            int cnt = count_bits(m);
            if (cnt == 0) return 0;
            if (cnt < bestCnt) {
                bestCnt = cnt;
                bestR = r;
                bestC = c;
                bestMask = m;
            }
        }
    }
    if (bestR == -1) return 1;

    int b = (bestR / 3) * 3 + bestC / 3;
    for (int d = 1; d <= 9; d++) {
        int bit = 1 << (d - 1);
        if (!(bestMask & bit)) continue;
        board[bestR][bestC] = (char)('0' + d);
        rowMask[bestR] |= bit;
        colMask[bestC] |= bit;
        boxMask[b] |= bit;
        if (dfs()) return 1;
        board[bestR][bestC] = '0';
        rowMask[bestR] &= ~bit;
        colMask[bestC] &= ~bit;
        boxMask[b] &= ~bit;
    }
    return 0;
}

int main(void) {
    for (int r = 0; r < 9; r++) {
        if (scanf("%9s", board[r]) != 1) return 0;
        for (int c = 0; c < 9; c++) {
            if (board[r][c] < '1' || board[r][c] > '9') continue;
            int bit = 1 << (board[r][c] - '1');
            int b = (r / 3) * 3 + c / 3;
            rowMask[r] |= bit;
            colMask[c] |= bit;
            boxMask[b] |= bit;
        }
    }
    dfs();
    for (int r = 0; r < 9; r++) printf("%s\n", board[r]);
    return 0;
}
```

### Python

```python
import sys

FULL = (1 << 9) - 1

def main():
    board = [list(sys.stdin.buffer.readline().decode().strip()) for _ in range(9)]
    row_mask = [0] * 9
    col_mask = [0] * 9
    box_mask = [0] * 9
    # 用三个位掩码维护行、列、宫的已用数字
    for r in range(9):
        for c in range(9):
            if board[r][c] != '0':
                bit = 1 << (ord(board[r][c]) - ord('1'))
                b = (r // 3) * 3 + c // 3
                row_mask[r] |= bit
                col_mask[c] |= bit
                box_mask[b] |= bit

    def dfs():
        best = None
        best_mask = 0
        best_cnt = 10
        for r in range(9):
            for c in range(9):
                if board[r][c] != '0':
                    continue
                b = (r // 3) * 3 + c // 3
                m = FULL & ~(row_mask[r] | col_mask[c] | box_mask[b])
                cnt = bin(m).count("1")
                if cnt == 0:
                    return False
                if cnt < best_cnt:
                    best_cnt = cnt
                    best = (r, c)
                    best_mask = m
        if best is None:
            return True
        r, c = best
        b = (r // 3) * 3 + c // 3
        while best_mask:
            bit = best_mask & -best_mask
            best_mask -= bit
            board[r][c] = chr(ord('1') + bit.bit_length() - 1)
            row_mask[r] |= bit
            col_mask[c] |= bit
            box_mask[b] |= bit
            if dfs():
                return True
            board[r][c] = '0'
            row_mask[r] &= ~bit
            col_mask[c] &= ~bit
            box_mask[b] &= ~bit
        return False

    dfs()
    sys.stdout.write("\n".join("".join(row) for row in board) + "\n")

main()
```
