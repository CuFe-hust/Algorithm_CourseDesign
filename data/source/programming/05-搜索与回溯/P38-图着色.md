# P38 图着色

- **标签**：#搜索与回溯 #图着色
- **难度**：★★★★☆
- **来源**：经典算法题（m 着色问题）

## 题目描述

给定一个包含 n 个结点的无向图，结点编号为 1 到 n。现在要用至多 k 种颜色给每个结点染上一种颜色，使得任意一条边的两个端点颜色都不同。

颜色不必全部使用。请判断是否存在满足条件的染色方案。

## 输入格式

第一行包含三个整数 n、m、k，分别表示结点数、边数和允许使用的颜色种数上界。

接下来 m 行，每行包含两个整数 u 和 v，表示结点 u 与结点 v 之间有一条无向边。保证 1 <= u < v <= n，且没有重边和自环。

## 输出格式

如果存在合法染色方案，输出一行 `Yes`；否则输出一行 `No`。输出区分大小写，行末需要换行。

## 样例输入

```text
4 4 3
1 2
1 3
2 3
3 4
```

## 样例输出

```text
Yes
```

## 数据范围

1 <= n <= 100，0 <= m <= n(n-1)/2，1 <= k <= n。图中没有重边和自环。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdbool.h>

int n, m, k;
bool adj[105][105];
int color[105];

// DSATUR 选点，颜色编号对称，只尝试已有颜色和一个新颜色
bool dfs(int colored, int colorCount) {
    if (colored == n) return true;

    int best = -1, bestSat = -1, bestDeg = -1;
    for (int v = 1; v <= n; v++) {
        if (color[v] != -1) continue;
        bool seen[105] = {false};
        int sat = 0, deg = 0;
        for (int u = 1; u <= n; u++) {
            if (!adj[v][u]) continue;
            if (color[u] != -1 && !seen[color[u]]) {
                seen[color[u]] = true;
                sat++;
            }
            if (color[u] == -1) deg++;
        }
        if (sat > bestSat || (sat == bestSat && deg > bestDeg)) {
            bestSat = sat;
            bestDeg = deg;
            best = v;
        }
    }

    int limit = colorCount < k ? colorCount : k - 1;
    for (int c = 0; c <= limit; c++) {
        bool ok = true;
        for (int u = 1; u <= n; u++) {
            if (adj[best][u] && color[u] == c) {
                ok = false;
                break;
            }
        }
        if (!ok) continue;
        color[best] = c;
        int nextCount = colorCount + (c == colorCount ? 1 : 0);
        if (dfs(colored + 1, nextCount)) return true;
        color[best] = -1;
    }
    return false;
}

int main(void) {
    if (scanf("%d %d %d", &n, &m, &k) != 3) return 0;
    for (int i = 0; i < m; i++) {
        int u, v;
        scanf("%d %d", &u, &v);
        adj[u][v] = adj[v][u] = true;
    }
    for (int i = 1; i <= n; i++) color[i] = -1;
    printf(dfs(0, 0) ? "Yes\n" : "No\n");
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
using namespace std;

int n, m, k;

// 优先选择饱和度高、剩余度大的结点染色
bool dfs(int colored, int colorCount) {
    if (colored == n) return true;

    int best = -1, bestSat = -1, bestDeg = -1;
    for (int v = 1; v <= n; ++v) {
        if (color[v] != -1) continue;
        vector<bool> seen(n + 1, false);
        int sat = 0, deg = 0;
        for (int u = 1; u <= n; ++u) {
            if (!adj[v][u]) continue;
            if (color[u] != -1 && !seen[color[u]]) {
                seen[color[u]] = true;
                ++sat;
            }
            if (color[u] == -1) ++deg;
        }
        if (sat > bestSat || (sat == bestSat && deg > bestDeg)) {
            bestSat = sat;
            bestDeg = deg;
            best = v;
        }
    }

    int limit = colorCount < k ? colorCount : k - 1;
    for (int c = 0; c <= limit; ++c) {
        bool ok = true;
        for (int u = 1; u <= n; ++u) {
            if (adj[best][u] && color[u] == c) {
                ok = false;
                break;
            }
        }
        if (!ok) continue;
        color[best] = c;
        int nextCount = colorCount + (c == colorCount ? 1 : 0);
        if (dfs(colored + 1, nextCount)) return true;
        color[best] = -1;
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> k;
    adj.assign(n + 1, vector<bool>(n + 1, false));
    color.assign(n + 1, -1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u][v] = adj[v][u] = true;

    cout << (dfs(0, 0) ? "Yes" : "No") << '\n';
    return 0;
}
```

### Python

```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m, k = data[0], data[1], data[2]
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    p = 3
    for _ in range(m):
        u, v = data[p], data[p + 1]
        p += 2
        adj[u][v] = adj[v][u] = True
    color = [-1] * (n + 1)

    # 颜色编号可互换，每一层只多尝试一种新颜色
    def dfs(colored, color_count):
        if colored == n:
            return True
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(1, n + 1):
            if color[v] != -1:
                continue
            seen = set()
            sat = 0
            deg = 0
            for u in range(1, n + 1):
                if not adj[v][u]:
                    continue
                if color[u] != -1 and color[u] not in seen:
                    seen.add(color[u])
                    sat += 1
                if color[u] == -1:
                    deg += 1
            if sat > best_sat or (sat == best_sat and deg > best_deg):
                best_sat, best_deg, best = sat, deg, v

        limit = color_count if color_count < k else k - 1
        for c in range(limit + 1):
            ok = True
            for u in range(1, n + 1):
                if adj[best][u] and color[u] == c:
                    ok = False
                    break
            if not ok:
                continue
            color[best] = c
            next_count = color_count + (1 if c == color_count else 0)
            if dfs(colored + 1, next_count):
                return True
            color[best] = -1
        return False

    sys.stdout.write("Yes\n" if dfs(0, 0) else "No\n")

main()
```
