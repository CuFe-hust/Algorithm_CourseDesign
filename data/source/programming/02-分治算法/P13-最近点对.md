# P13 最近点对

- **标签**：#分治算法 #计算几何
- **难度**：★★★★☆
- **来源**：经典算法题（Closest Pair of Points）

## 题目描述

平面上给定 n 个点，第 i 个点的坐标为 (x_i, y_i)。两个点 (x_1, y_1) 与 (x_2, y_2) 之间的欧几里得距离定义为

sqrt((x_1 - x_2)^2 + (y_1 - y_2)^2)

请你求出所有点对中距离的最小值。同一位置允许存在多个点；如果两个点重合，它们之间的距离为 0。

本题要求使用分治算法完成，时间复杂度为 O(n log n)。

## 输入格式

输入共 n + 1 行：

- 第一行包含一个整数 n。
- 接下来 n 行，每行包含两个整数 x_i 和 y_i，中间用一个空格分隔，表示第 i 个点的坐标。

输入只有一组测试数据。

## 输出格式

输出一行一个实数，表示最近两点间的距离，四舍五入保留 4 位小数。

即使最小距离是整数，也必须输出 4 位小数，例如最小距离为 5 时应输出 5.0000。

## 样例输入
```text
2
0 0
3 4
```

## 样例输出
```text
5.0000
```

## 数据范围

- 2 <= n <= 10^5
- |x_i| <= 10^9
- |y_i| <= 10^9

答案非负，且固定保留 4 位小数。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <limits.h>

typedef struct {
    long long x, y;
} Point;

static Point p[100005];
static int idx[100005], tmp[100005];

static int cmp_x(const void *a, const void *b) {
    const Point *u = (const Point *)a, *v = (const Point *)b;
    if (u->x != v->x) return u->x < v->x ? -1 : 1;
    if (u->y != v->y) return u->y < v->y ? -1 : 1;
    return 0;
}

/* 分治求最近点对，返回最小距离的平方；idx[l, r) 始终按 y 升序 */
static long long closest(int l, int r) {
    if (r - l <= 3) {
        long long best = LLONG_MAX;
        for (int i = l; i < r; i++) {
            for (int j = i + 1; j < r; j++) {
                long long dx = p[i].x - p[j].x;
                long long dy = p[i].y - p[j].y;
                long long d2 = dx * dx + dy * dy;
                if (d2 < best) best = d2;
            }
        }
        for (int i = l + 1; i < r; i++) { /* 按 y 插入排序 */
            int v = idx[i], j = i - 1;
            while (j >= l && p[idx[j]].y > p[v].y) {
                idx[j + 1] = idx[j];
                j--;
            }
            idx[j + 1] = v;
        }
        return best;
    }
    int m = (l + r) / 2;
    long long midx = p[m].x;
    long long d1 = closest(l, m), d2 = closest(m, r);
    long long d = d1 < d2 ? d1 : d2;
    int i = l, j = m, k = l; /* 归并两段按 y 有序的下标 */
    while (i < m && j < r) {
        if (p[idx[i]].y <= p[idx[j]].y) tmp[k++] = idx[i++];
        else tmp[k++] = idx[j++];
    }
    while (i < m) tmp[k++] = idx[i++];
    while (j < r) tmp[k++] = idx[j++];
    for (i = l; i < r; i++) idx[i] = tmp[i];
    if (d == 0) return 0;
    for (i = l; i < r; i++) {
        long long xi = p[idx[i]].x, yi = p[idx[i]].y;
        long long dxm = xi - midx;
        if (dxm * dxm >= d) continue;
        for (j = i + 1; j < r; j++) {
            long long dy = p[idx[j]].y - yi;
            if (dy * dy >= d) break;
            long long dx = p[idx[j]].x - xi;
            long long d2 = dx * dx + dy * dy;
            if (d2 < d) d = d2;
        }
    }
    return d;
}

int main(void) {
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lld %lld", &p[i].x, &p[i].y);
        idx[i] = i;
    }
    qsort(p, n, sizeof(Point), cmp_x);
    for (int i = 0; i < n; i++) idx[i] = i;
    long long best = closest(0, n);
    printf("%.4f\n", sqrt((double)best));
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <iomanip>
#include <climits>
using namespace std;

struct Point {
    long long x, y;
};

vector<Point> p;
vector<int> idx, tmp;

/* 分治求最近点对，返回最小距离的平方；idx[l, r) 始终按 y 升序 */
long long closest(int l, int r) {
    if (r - l <= 3) {
        long long best = LLONG_MAX;
        for (int i = l; i < r; i++) {
            for (int j = i + 1; j < r; j++) {
                long long dx = p[i].x - p[j].x;
                long long dy = p[i].y - p[j].y;
                best = min(best, dx * dx + dy * dy);
            }
        }
        sort(idx.begin() + l, idx.begin() + r,
             [](int a, int b) { return p[a].y < p[b].y; }); /* 按 y 排序 */
        return best;
    }
    int m = (l + r) / 2;
    long long midx = p[m].x;
    long long d = min(closest(l, m), closest(m, r));
    int i = l, j = m, k = l; /* 归并两段按 y 有序的下标 */
    while (i < m && j < r) {
        if (p[idx[i]].y <= p[idx[j]].y) tmp[k++] = idx[i++];
        else tmp[k++] = idx[j++];
    }
    while (i < m) tmp[k++] = idx[i++];
    while (j < r) tmp[k++] = idx[j++];
    for (i = l; i < r; i++) idx[i] = tmp[i];
    if (d == 0) return 0;
    for (i = l; i < r; i++) {
        long long xi = p[idx[i]].x, yi = p[idx[i]].y;
        long long dxm = xi - midx;
        if (dxm * dxm >= d) continue;
        for (j = i + 1; j < r; j++) {
            long long dy = p[idx[j]].y - yi;
            if (dy * dy >= d) break;
            long long dx = p[idx[j]].x - xi;
            long long d2 = dx * dx + dy * dy;
            if (d2 < d) d = d2;
        }
    }
    return d;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    p.resize(n);
    idx.resize(n);
    tmp.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i].x >> p[i].y;
        idx[i] = i;
    }
    sort(p.begin(), p.end(), [](const Point &a, const Point &b) {
        return a.x != b.x ? a.x < b.x : a.y < b.y;
    });
    for (int i = 0; i < n; i++) idx[i] = i;
    long long best = closest(0, n);
    cout << fixed << setprecision(4) << sqrt((double)best) << '\n';
    return 0;
}
```

### Python

```python
import sys
import math


def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    pts = []
    for i in range(n):
        pts.append((int(data[1 + 2 * i]), int(data[2 + 2 * i])))
    pts.sort()
    py = [q[1] for q in pts]
    idx = list(range(n))
    tmp = [0] * n
    sys.setrecursionlimit(300000)

    # 分治求最近点对，返回最小距离的平方；idx[l, r) 始终按 y 升序
    def closest(l, r):
        if r - l <= 3:
            best = None
            for i in range(l, r):
                xi, yi = pts[i]
                for j in range(i + 1, r):
                    dx = xi - pts[j][0]
                    dy = yi - pts[j][1]
                    d2 = dx * dx + dy * dy
                    if best is None or d2 < best:
                        best = d2
            idx[l:r] = sorted(idx[l:r], key=py.__getitem__)
            return best if best is not None else 1 << 128
        m = (l + r) // 2
        midx = pts[m][0]
        d = min(closest(l, m), closest(m, r))
        i, j, k = l, m, l
        while i < m and j < r:
            if py[idx[i]] <= py[idx[j]]:
                tmp[k] = idx[i]
                i += 1
            else:
                tmp[k] = idx[j]
                j += 1
            k += 1
        while i < m:
            tmp[k] = idx[i]
            i += 1
            k += 1
        while j < r:
            tmp[k] = idx[j]
            j += 1
            k += 1
        idx[l:r] = tmp[l:r]
        if d == 0:
            return 0
        for i in range(l, r):
            xi = pts[idx[i]][0]
            yi = pts[idx[i]][1]
            dxm = xi - midx
            if dxm * dxm >= d:
                continue
            for j in range(i + 1, r):
                dy = pts[idx[j]][1] - yi
                if dy * dy >= d:
                    break
                dx = pts[idx[j]][0] - xi
                d2 = dx * dx + dy * dy
                if d2 < d:
                    d = d2
        return d

    best = closest(0, n)
    print(f"{math.sqrt(best):.4f}")


main()
```
