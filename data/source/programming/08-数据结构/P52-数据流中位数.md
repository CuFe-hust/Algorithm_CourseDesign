# P52 数据流中位数

- **标签**：#数据结构 #堆
- **难度**：★★★★☆
- **来源**：LeetCode 295 变体

## 题目描述

给定一个长度为 n 的整数序列 a_1, a_2, ..., a_n。按照输入顺序依次处理这些数，对于每个前缀 a_1, a_2, ..., a_i（1 <= i <= n），求出该前缀的中位数。

本题中位数的定义是：将当前前缀的 i 个数按非递减顺序排列后，位于第 ceil(i/2) 个位置的数，其中位置从 1 开始编号。也就是说，i 为奇数时取正中间的数，i 为偶数时取偏小的那个中间数。因此中位数一定是序列中出现过的整数，不需要输出小数。

例如，对于前缀 4, 1, 3, 2，排序后为 1, 2, 3, 4，i = 4，取第 2 个数，中位数为 2。

## 输入格式

输入只有一组测试数据。

第一行包含一个整数 n（1 <= n <= 10^5）。

第二行包含 n 个整数 a_1, a_2, ..., a_n（|a_i| <= 10^9），相邻整数之间用一个空格分隔。

## 输出格式

输出一行 n 个整数，第 i 个整数表示前缀 a_1, a_2, ..., a_i 的中位数。相邻整数之间用一个空格分隔，行末不得有多余空格，输出末尾换行。

## 样例输入

```text
5
1 2 3 4 5
```

## 样例输出

```text
1 1 2 2 3
```

## 数据范围

- 1 <= n <= 10^5
- |a_i| <= 10^9
- 输入只有一组测试数据
- 所有中位数均在 32 位有符号整数范围内

## 标准实现

### C

```c
#include <stdio.h>

#define MAXN 100005

long long low[MAXN], high[MAXN];
int lowSize = 0, highSize = 0;

static void swapLL(long long *x, long long *y) {
    long long t = *x;
    *x = *y;
    *y = t;
}

static void pushLow(long long x) {
    int i = lowSize++;
    low[i] = x;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (low[p] >= low[i]) break;
        swapLL(&low[p], &low[i]);
        i = p;
    }
}

static long long popLow(void) {
    long long res = low[0];
    low[0] = low[--lowSize];
    int i = 0;
    while (1) {
        int l = i * 2 + 1, r = l + 1, best = i;
        if (l < lowSize && low[l] > low[best]) best = l;
        if (r < lowSize && low[r] > low[best]) best = r;
        if (best == i) break;
        swapLL(&low[i], &low[best]);
        i = best;
    }
    return res;
}

static void pushHigh(long long x) {
    int i = highSize++;
    high[i] = x;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (high[p] <= high[i]) break;
        swapLL(&high[p], &high[i]);
        i = p;
    }
}

static long long popHigh(void) {
    long long res = high[0];
    high[0] = high[--highSize];
    int i = 0;
    while (1) {
        int l = i * 2 + 1, r = l + 1, best = i;
        if (l < highSize && high[l] < high[best]) best = l;
        if (r < highSize && high[r] < high[best]) best = r;
        if (best == i) break;
        swapLL(&high[i], &high[best]);
        i = best;
    }
    return res;
}

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) return 0;

    // 大根堆 low 保存较小的一半，小根堆 high 保存较大的一半
    for (int i = 0; i < n; i++) {
        long long x;
        if (scanf("%lld", &x) != 1) return 0;
        pushLow(x);
        pushHigh(popLow());
        if (highSize > lowSize) {
            pushLow(popHigh());
        }
        if (i > 0) putchar(' ');
        printf("%lld", low[0]);
    }
    putchar('\n');
    return 0;
}
```

### C++

```cpp
#include <functional>
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    // low 为大根堆，high 为小根堆；平衡后 low 的元素个数不少于 high
    priority_queue<long long> low;
    priority_queue<long long, vector<long long>, greater<long long>> high;

    for (int i = 0; i < n; ++i) {
        long long x;
        cin >> x;
        low.push(x);
        high.push(low.top());
        low.pop();
        if (high.size() > low.size()) {
            low.push(high.top());
            high.pop();
        }
        if (i) cout << ' ';
        cout << low.top();
    }
    cout << '\n';
    return 0;
}
```

### Python

```python
import heapq
import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n = data[0]
    a = data[1:1 + n]

    low = []   # 存负数，模拟大根堆
    high = []  # 小根堆
    out = []
    for x in a:
        heapq.heappush(low, -x)
        heapq.heappush(high, -heapq.heappop(low))
        if len(high) > len(low):
            heapq.heappush(low, -heapq.heappop(high))
        out.append(str(-low[0]))

    sys.stdout.write(" ".join(out) + "\n")


if __name__ == "__main__":
    main()
```
