# P47 快速排序

- **标签**：#排序 #快速排序 #分治算法
- **难度**：★★☆☆☆
- **来源**：经典算法题（Quicksort）

## 题目描述

给定 n 个整数，请将它们按从小到大的顺序排列后输出。

本题要求你自行实现快速排序算法完成排序，不得调用语言自带的排序函数（例如 C 的 qsort、C++ 的 std::sort、Python 的 sorted 或 list.sort）。

序列中可能含有相同的整数，排序后这些相同的整数应当连续出现。

提示：本题数据中可能出现大量重复元素，也可能已经基本有序，请保证你实现的快速排序在这些数据上仍然高效。

## 输入格式

第一行一个整数 n，表示整数的个数。

第二行包含 n 个整数 a_1, a_2, …, a_n，相邻两个整数之间用一个空格分隔。

## 输出格式

输出一行，包含 n 个整数，为从小到大排好序的序列，相邻两个整数之间用一个空格分隔。行末不得有多余空格。

## 样例输入

```text
5
3 1 4 1 5
```

## 样例输出

```text
1 1 3 4 5
```

## 数据范围

1 ≤ n ≤ 10^5，|a_i| ≤ 10^9。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

static int a[100005];

static void swap(int *x, int *y) {
    int t = *x;
    *x = *y;
    *y = t;
}

/* 随机化三路划分快速排序：小于/等于/大于基准值 */
static void qsort3(int lo, int hi) {
    while (lo < hi) {
        int r = lo + rand() % (hi - lo + 1);
        swap(&a[r], &a[lo]);            /* 随机选取基准值 */
        int pivot = a[lo];
        int lt = lo, i = lo + 1, gt = hi;
        while (i <= gt) {
            if (a[i] < pivot) swap(&a[lt++], &a[i++]);
            else if (a[i] > pivot) swap(&a[i], &a[gt--]);
            else i++;
        }
        if (lt - lo < hi - gt) {
            qsort3(lo, lt - 1);
            lo = gt + 1;
        } else {
            qsort3(gt + 1, hi);
            hi = lt - 1;
        }
    }
}

int main(void) {
    srand((unsigned)time(NULL));
    int n;
    if (scanf("%d", &n) != 1) return 0;
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    qsort3(0, n - 1);
    for (int i = 0; i < n; i++) {
        if (i) putchar(' ');
        printf("%d", a[i]);
    }
    putchar('\n');
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <cstdlib>
#include <ctime>
using namespace std;

static void qsort3(vector<int> &a, int lo, int hi) {
    while (lo < hi) {
        int r = lo + rand() % (hi - lo + 1);
        swap(a[r], a[lo]);                 // 随机选取基准值
        int pivot = a[lo], lt = lo, i = lo + 1, gt = hi;
        while (i <= gt) {
            if (a[i] < pivot) swap(a[lt++], a[i++]);
            else if (a[i] > pivot) swap(a[i], a[gt--]);
            else ++i;
        }
        if (lt - lo < hi - gt) {           // 先递归较小一侧，控制栈深度
            qsort3(a, lo, lt - 1);
            lo = gt + 1;
        } else {
            qsort3(a, gt + 1, hi);
            hi = lt - 1;
        }
    }
}

int main() {
    srand((unsigned)time(nullptr));
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    qsort3(a, 0, n - 1);
    for (int i = 0; i < n; ++i) cout << (i ? " " : "") << a[i];
    cout << '\n';
    return 0;
}
```

### Python

```python
import sys
import random


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n = data[0]
    a = data[1:1 + n]
    # 随机化三路划分快速排序，手工栈避免递归过深
    stack = [(0, n - 1)]
    while stack:
        lo, hi = stack.pop()
        while lo < hi:
            r = random.randrange(lo, hi + 1)
            a[r], a[lo] = a[lo], a[r]   # 随机选取基准值
            pivot = a[lo]
            lt, i, gt = lo, lo + 1, hi
            while i <= gt:
                if a[i] < pivot:
                    a[lt], a[i] = a[i], a[lt]
                    lt += 1
                    i += 1
                elif a[i] > pivot:
                    a[i], a[gt] = a[gt], a[i]
                    gt -= 1
                else:
                    i += 1
            # 较小一侧入栈，较大一侧继续循环
            if lt - lo < hi - gt:
                if lo < lt - 1:
                    stack.append((lo, lt - 1))
                lo = gt + 1
            else:
                if gt + 1 < hi:
                    stack.append((gt + 1, hi))
                hi = lt - 1
    sys.stdout.write(' '.join(map(str, a)) + '\n')


main()
```
