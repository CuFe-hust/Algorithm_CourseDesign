# P43 前缀统计

- **标签**：#字符串 #字典树
- **难度**：★★★☆☆
- **来源**：经典算法题（Trie 前缀统计）

## 题目描述

给定 n 个仅由小写字母组成的单词，再给出 q 次询问。每次询问给出一个字符串 t，请统计这 n 个单词中有多少个单词以 t 作为前缀。一个单词与 t 完全相同也算以 t 为前缀。

注意：n 个单词中可能出现相同的单词，此时每个单词都单独计数。

## 输入格式

第一行一个整数 n（1 <= n <= 10^5）。

接下来 n 行，每行一个仅由小写字母组成的单词。所有单词的总长度不超过 10^5。

接下来一行一个整数 q（1 <= q <= 10^5）。

接下来 q 行，每行一个仅由小写字母组成的询问字符串 t。

## 输出格式

输出 q 行，每行一个整数，依次表示每次询问的答案，即有多少个单词以询问字符串为前缀。若没有任何单词以该字符串为前缀，输出 0。

## 样例输入

```text
5
apple
app
apply
ban
banana
3
app
b
z
```

## 样例输出

```text
3
2
0
```

## 数据范围

- 1 <= n <= 10^5
- 所有单词的总长度不超过 10^5
- 1 <= q <= 10^5
- 所有询问字符串的总长度不超过 10^5
- 所有的单词和询问字符串均只含小写字母，且长度至少为 1
- 相同的单词分别计数，例如单词 a 出现两次时，询问 a 的答案为 2

## 标准实现

### C

```c
#include <stdio.h>

#define MAXNODE 100005

/* 字典树：child 存儿子编号，cnt 存经过该结点的单词数 */
static int child[MAXNODE][26];
static int cnt[MAXNODE];
static char buf[MAXNODE];

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) return 0;
    int nodes = 1;

    for (int i = 0; i < n; i++) {
        if (scanf("%s", buf) != 1) return 0;
        int u = 0;
        for (char *q = buf; *q; q++) {
            int c = *q - 'a';
            if (child[u][c] == 0) child[u][c] = nodes++;
            u = child[u][c];
            cnt[u]++;
        }
    }

    int qn;
    if (scanf("%d", &qn) != 1) return 0;
    while (qn--) {
        if (scanf("%s", buf) != 1) return 0;
        int u = 0, ok = 1;
        for (char *q = buf; *q; q++) {
            int c = *q - 'a';
            if (child[u][c] == 0) {
                ok = 0;
                break;
            }
            u = child[u][c];
        }
        printf("%d\n", ok ? cnt[u] : 0);
    }
    return 0;
}
```

### C++

```cpp
#include <array>
#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (!(cin >> n)) return 0;

    // 字典树：child 存儿子编号，cnt 存经过该结点的单词数
    vector<array<int, 26>> child;
    vector<int> cnt;
    child.reserve(100005);
    cnt.reserve(100005);
    child.push_back({});
    child[0].fill(0);
    cnt.push_back(0);

    for (int i = 0; i < n; ++i) {
        string w;
        cin >> w;
        int u = 0;
        for (char ch : w) {
            int c = ch - 'a';
            if (child[u][c] == 0) {
                child[u][c] = (int)child.size();
                child.push_back({});
                child.back().fill(0);
                cnt.push_back(0);
            }
            u = child[u][c];
            ++cnt[u];
        }
    }

    int q;
    cin >> q;
    while (q--) {
        string w;
        cin >> w;
        int u = 0;
        bool ok = true;
        for (char ch : w) {
            int c = ch - 'a';
            if (child[u][c] == 0) {
                ok = false;
                break;
            }
            u = child[u][c];
        }
        cout << (ok ? cnt[u] : 0) << '\n';
    }
    return 0;
}
```

### Python

```python
import sys


def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    pos = 0
    n = int(data[pos])
    pos += 1

    # 字典树：child 用字典存儿子，cnt 存经过该结点的单词数
    child = [{}]
    cnt = [0]
    for _ in range(n):
        w = data[pos]
        pos += 1
        u = 0
        for c in w:
            v = child[u].get(c)
            if v is None:
                v = len(child)
                child[u][c] = v
                child.append({})
                cnt.append(0)
            u = v
            cnt[u] += 1

    q = int(data[pos])
    pos += 1
    out = []
    for _ in range(q):
        w = data[pos]
        pos += 1
        u = 0
        ok = True
        for c in w:
            v = child[u].get(c)
            if v is None:
                ok = False
                break
            u = v
        out.append(str(cnt[u]) if ok else "0")

    sys.stdout.write("\n".join(out) + "\n")


if __name__ == "__main__":
    main()
```
