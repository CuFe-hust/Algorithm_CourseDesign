# P42 最长回文子串

- **标签**：#字符串 #Manacher #动态规划
- **难度**：★★★★☆
- **来源**：LeetCode 5 / Manacher

## 题目描述

如果一个字符串正着读和倒着读完全相同，则称它为回文串。给定一个只含小写字母的字符串 s，请求出 s 的最长回文子串。

子串是指 s 中一段连续字符组成的字符串。若长度最长的回文子串有多个，输出起始下标最小的那一个。字符串下标从 0 开始。

## 输入格式

输入只有一行，一个仅由小写英文字母组成的字符串 s，长度满足 1 <= |s| <= 10^5。

## 输出格式

输出共两行。

第一行是一个整数，表示最长回文子串的长度。

第二行是一个字符串，表示该最长回文子串本身。若存在多个长度相同的最长回文子串，输出起始下标最小的那一个。输出末尾换行。

## 样例输入

```text
babad
```

## 样例输出

```text
3
bab
```

## 数据范围

- 1 <= |s| <= 10^5
- s 仅由小写英文字母 a 到 z 组成
- 单个字符本身也是回文串，因此答案长度至少为 1
- 若有多个等长的最长回文子串，取起始下标最小者

## 标准实现

### C

```c
#include <stdio.h>
#include <string.h>

#define MAXN 200005

static char s[100005];
static char t[MAXN];
static int rad[MAXN];

int main(void) {
    if (scanf("%s", s) != 1) return 0;
    int n = (int)strlen(s);
    int m = 2 * n + 1;
    for (int i = 0; i < n; i++) {
        t[2 * i] = '#';
        t[2 * i + 1] = s[i];
    }
    t[2 * n] = '#';

    /* 插入分隔符后使用 Manacher 求各中心回文半径 */
    int center = 0, right = 0;
    int bestLen = 0, bestStart = 0;
    for (int i = 0; i < m; i++) {
        int r = 0;
        if (i < right) {
            int mirror = 2 * center - i;
            r = rad[mirror] < right - i ? rad[mirror] : right - i;
        }
        while (i - r - 1 >= 0 && i + r + 1 < m && t[i - r - 1] == t[i + r + 1]) r++;
        rad[i] = r;
        if (i + r > right) {
            center = i;
            right = i + r;
        }
        if (r > bestLen) {
            bestLen = r;
            bestStart = (i - r) / 2;
        }
    }

    printf("%d\n", bestLen);
    for (int k = 0; k < bestLen; k++) putchar(s[bestStart + k]);
    putchar('\n');
    return 0;
}
```

```cpp
#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string s;
    if (!(cin >> s)) return 0;
    int n = (int)s.size();
    int m = 2 * n + 1;

    string t(m, '#');
    for (int i = 0; i < n; ++i) t[2 * i + 1] = s[i];

    // 插入分隔符后使用 Manacher 求各中心回文半径
    vector<int> rad(m, 0);
    int center = 0, right = 0;
    int bestLen = 0, bestStart = 0;
    for (int i = 0; i < m; ++i) {
        int r = 0;
        if (i < right) {
            int mirror = 2 * center - i;
            r = min(rad[mirror], right - i);
        }
        while (i - r - 1 >= 0 && i + r + 1 < m && t[i - r - 1] == t[i + r + 1]) ++r;
        rad[i] = r;
        if (i + r > right) {
            center = i;
            right = i + r;
        }
        if (r > bestLen) {
            bestLen = r;
            bestStart = (i - r) / 2;
        }
    }

    cout << bestLen << '\n';
    cout << s.substr(bestStart, bestLen) << '\n';
    return 0;
}
```

### Python

```python
import sys


def main():
    s = sys.stdin.buffer.readline().strip().decode()
    if not s:
        return
    n = len(s)

    # 插入分隔符后使用 Manacher 求各中心回文半径
    t = ['#'] * (2 * n + 1)
    for i in range(n):
        t[2 * i + 1] = s[i]

    m = 2 * n + 1
    rad = [0] * m
    center = 0
    right = 0
    best_len = 0
    best_start = 0
    for i in range(m):
        r = 0
        if i < right:
            r = min(rad[2 * center - i], right - i)
        while i - r - 1 >= 0 and i + r + 1 < m and t[i - r - 1] == t[i + r + 1]:
            r += 1
        rad[i] = r
        if i + r > right:
            center = i
            right = i + r
        if r > best_len:
            best_len = r
            best_start = (i - r) // 2

    sys.stdout.write(str(best_len) + "\n")
    sys.stdout.write(s[best_start:best_start + best_len] + "\n")


if __name__ == "__main__":
    main()
```
