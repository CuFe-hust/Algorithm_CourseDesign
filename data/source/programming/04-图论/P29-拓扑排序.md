# P29 拓扑排序

- **标签**：#图论 #拓扑排序
- **难度**：★★☆☆☆
- **来源**：经典算法题（Topological Sort）

## 题目描述

有 \(n\) 个任务，编号为 \(1\) 到 \(n\)，另有 \(m\) 条先后关系约束。每条约束用两个整数 \(u\)、\(v\) 表示，含义是任务 \(u\) 必须在任务 \(v\) 之前完成。

请找出一个合法的拓扑序列，使得每条约束中的 \(u\) 都排在 \(v\) 之前，并且在所有合法拓扑序列中字典序最小。比较两个序列的字典序时，从左到右找到第一个不同的位置，该位置任务编号较小的序列更小。

若图中存在环，则不存在合法的拓扑序列，输出 `-1`。

## 输入格式

第一行包含两个整数 \(n\) 和 \(m\)，分别表示任务数和先后关系数。

接下来 \(m\) 行，每行包含两个整数 \(u\)、\(v\)，表示任务 \(u\) 必须在任务 \(v\) 之前完成。

本题只有一组测试数据。输入保证不含重边和自环，但可能含有环。

## 输出格式

输出一行，包含字典序最小的合法拓扑序列，序列中的任务编号用一个空格分隔。

若图中存在环，输出 `-1`。

## 样例输入

```text
4 3
1 2
1 3
3 4
```

## 样例输出

```text
1 2 3 4
```

## 数据范围

- \(1 <= n <= 10^5\)，\(0 <= m <= 2 x 10^5\)。
- \(1 <= u,v <= n\)。
- 输入保证不含重边和自环，但可能含有环。
- 若存在合法序列，其长度一定为 \(n\)，且所有任务编号恰好出现一次。

## 标准实现

### C

```c
#include <stdio.h>

static int head[100005], to[200005], nxt[200005], indeg[100005];
static int heap[100005], hsize = 0;

static void push(int x) {
    int i = ++hsize;
    heap[i] = x;
    while (i > 1) {
        int p = i / 2;
        if (heap[p] <= heap[i]) break;
        int t = heap[p];
        heap[p] = heap[i];
        heap[i] = t;
        i = p;
    }
}

static int pop(void) {
    int res = heap[1];
    heap[1] = heap[hsize--];
    int i = 1;
    while (1) {
        int l = i * 2, r = l + 1, s = i;
        if (l <= hsize && heap[l] < heap[s]) s = l;
        if (r <= hsize && heap[r] < heap[s]) s = r;
        if (s == i) break;
        int t = heap[s];
        heap[s] = heap[i];
        heap[i] = t;
        i = s;
    }
    return res;
}

int main(void) {
    int n, m;
    if (scanf("%d %d", &n, &m) != 2) return 0;

    for (int i = 1; i <= n; i++) head[i] = -1;
    for (int i = 0; i < m; i++) {
        int u, v;
        scanf("%d %d", &u, &v);
        to[i] = v;
        nxt[i] = head[u];
        head[u] = i;
        indeg[v]++;
    }

    for (int i = 1; i <= n; i++) {
        if (indeg[i] == 0) push(i);
    }

    int cnt = 0;
    static int order[100005];
    // Kahn 算法配合小根堆，每一步取当前可选的最小结点
    while (hsize > 0) {
        int u = pop();
        order[cnt++] = u;
        for (int e = head[u]; e != -1; e = nxt[e]) {
            int v = to[e];
            indeg[v]--;
            if (indeg[v] == 0) push(v);
        }
    }

    if (cnt < n) {
        puts("-1");
    } else {
        for (int i = 0; i < cnt; i++) {
            if (i > 0) putchar(' ');
            printf("%d", order[i]);
        }
        putchar('\n');
    }
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <functional>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<vector<int>> adj(n + 1);
    vector<int> indeg(n + 1, 0);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        indeg[v]++;
    }

    priority_queue<int, vector<int>, greater<int>> pq;
    for (int i = 1; i <= n; i++) {
        if (indeg[i] == 0) pq.push(i);
    }

    vector<int> order;
    // Kahn 算法配合小根堆，保证每一步选择最小编号
    while (!pq.empty()) {
        int u = pq.top();
        pq.pop();
        order.push_back(u);
        for (int v : adj[u]) {
            if (--indeg[v] == 0) pq.push(v);
        }
    }

    if ((int)order.size() < n) {
        cout << -1 << '\n';
    } else {
        for (int i = 0; i < n; i++) {
            if (i > 0) cout << ' ';
            cout << order[i];
        }
        cout << '\n';
    }
    return 0;
}
```

### Python

```python
import sys
import heapq


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    p = 2
    for _ in range(m):
        u, v = data[p], data[p + 1]
        p += 2
        adj[u].append(v)
        indeg[v] += 1

    heap = [i for i in range(1, n + 1) if indeg[i] == 0]
    heapq.heapify(heap)
    order = []
    # Kahn 算法配合小根堆，保证每一步选择最小编号
    while heap:
        u = heapq.heappop(heap)
        order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                heapq.heappush(heap, v)

    if len(order) < n:
        print(-1)
    else:
        print(" ".join(map(str, order)))


if __name__ == "__main__":
    main()
```
