# P33 强连通分量

- **标签**：#图论 #强连通分量 #Tarjan
- **难度**：★★★★☆
- **来源**：经典算法题（SCC / Tarjan）

## 题目描述

给定一张有 n 个结点、m 条有向边的图，结点编号为 1,2,...,n。

如果两个结点 u 和 v 满足：既存在一条从 u 到 v 的有向路径，又存在一条从 v 到 u 的有向路径，则称 u 与 v 强连通。约定路径长度可以为 0，因此任何结点都与自身强连通。

有向图的一个强连通分量，指的是满足“集合内任意两个结点都强连通”的极大结点集合。单独一个结点（即使它不与任何其他结点相连）也单独构成一个强连通分量。

请统计这张有向图中强连通分量的个数。

图中可能存在重边（多条从 u 到 v 的边）和自环（u=v 的边）。

## 输入格式

第一行包含两个整数 n 和 m，分别表示结点数和有向边数。

接下来 m 行，每行包含两个整数 u 和 v（1 <= u,v <= n），表示一条从 u 指向 v 的有向边。

输入只有一组数据。

## 输出格式

输出一行一个整数，表示强连通分量的个数。

## 样例输入

```text
5 6
1 2
2 3
3 1
3 4
4 5
5 4
```

## 样例输出

```text
2
```

## 数据范围

- 1 <= n <= 10^4
- 0 <= m <= 5 x 10^4
- 1 <= u,v <= n
- 允许重边和自环。

## 标准实现

### C

```c
#include <stdio.h>
/* dfn：DFS 时间戳；low：能回溯到的最小时间戳 */

#define MAXN 10005
#define MAXM 50005

int head[MAXN], nxt[MAXM], to[MAXM], ecnt;
int dfn[MAXN], low[MAXN], inStack[MAXN], stk[MAXN];
int timer, top, sccCnt;

void addEdge(int u, int v) {
    to[++ecnt] = v;
    nxt[ecnt] = head[u];
    head[u] = ecnt;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk[++top] = u;
    inStack[u] = 1;
    for (int e = head[u]; e; e = nxt[e]) {
        int v = to[e];
        if (!dfn[v]) {
            tarjan(v);
            if (low[v] < low[u]) low[u] = low[v];
        } else if (inStack[v] && dfn[v] < low[u]) {
            low[u] = dfn[v];
        }
    }
    if (low[u] == dfn[u]) {
        ++sccCnt;
        int w;
        do {
            w = stk[top--];
            inStack[w] = 0;
        } while (w != u);
    }
}

int main(void) {
    int n, m;
    if (scanf("%d %d", &n, &m) != 2) return 0;
    for (int i = 0; i < m; ++i) {
        int u, v;
        scanf("%d %d", &u, &v);
        addEdge(u, v);
    }
    for (int i = 1; i <= n; ++i) {
        if (!dfn[i]) tarjan(i);
    }
    printf("%d\n", sccCnt);
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int n, m, timerCnt, sccCnt;
vector<vector<int>> g;
vector<int> dfn, low, stk;
vector<char> inStk;

// Tarjan 求强连通分量
void tarjan(int u) {
    dfn[u] = low[u] = ++timerCnt;
    stk.push_back(u);
    inStk[u] = 1;
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else if (inStk[v]) {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (low[u] == dfn[u]) {
        ++sccCnt;
        while (true) {
            int w = stk.back();
            stk.pop_back();
            inStk[w] = 0;
            if (w == u) break;
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    g.assign(n + 1, {});
    dfn.assign(n + 1, 0);
    low.assign(n + 1, 0);
    inStk.assign(n + 1, 0);
    // 邻接表存图
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }
    for (int i = 1; i <= n; ++i) {
        if (!dfn[i]) tarjan(i);
    }
    cout << sccCnt << '\n';
    return 0;
}
```

### Python

```python
import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n + 1)]
    pos = 2
    for _ in range(m):
        u, v = data[pos], data[pos + 1]
        pos += 2
        adj[u].append(v)

    dfn = [0] * (n + 1)
    low = [0] * (n + 1)
    on_stack = [False] * (n + 1)
    timer = 0
    scc_cnt = 0

    # 迭代版 Tarjan，栈中保存（结点，下一条待处理边的下标）
    for start in range(1, n + 1):
        if dfn[start]:
            continue
        dfs_stack = [(start, 0)]
        path = []
        while dfs_stack:
            u, edge_idx = dfs_stack.pop()
            if edge_idx == 0:
                timer += 1
                dfn[u] = low[u] = timer
                on_stack[u] = True
                path.append(u)

            if edge_idx < len(adj[u]):
                v = adj[u][edge_idx]
                dfs_stack.append((u, edge_idx + 1))
                if dfn[v] == 0:
                    dfs_stack.append((v, 0))
                elif on_stack[v] and dfn[v] < low[u]:
                    low[u] = dfn[v]
            else:
                if dfs_stack:
                    parent = dfs_stack[-1][0]
                    if low[u] < low[parent]:
                        low[parent] = low[u]
                # 找到一个新的强连通分量
                if low[u] == dfn[u]:
                    scc_cnt += 1
                    while True:
                        w = path.pop()
                        on_stack[w] = False
                        if w == u:
                            break

    sys.stdout.write(str(scc_cnt) + "\n")


main()
```
