# P26 单源最短路

- **标签**：#图论 #最短路 #Dijkstra
- **难度**：★★★☆☆
- **来源**：经典算法题（Dijkstra）

## 题目描述

给定一个包含 \(n\) 个结点、\(m\) 条边的有向图，结点编号为 \(1\) 到 \(n\)，每条边的边权都是非负整数。请求出从 \(1\) 号结点到 \(n\) 号结点的最短路径长度。路径长度定义为路径上所有边权之和。

图中允许重边和自环。若不存在从 \(1\) 号结点到 \(n\) 号结点的路径，输出 `-1`。从 \(1\) 号结点到它自身的距离为 \(0\)，因此当 \(n=1\) 时答案总是 `0`。

## 输入格式

第一行包含两个整数 \(n\) 和 \(m\)，分别表示结点数和边数。

接下来 \(m\) 行，每行包含三个整数 \(u\)、\(v\)、\(w\)，表示一条从 \(u\) 指向 \(v\)、边权为 \(w\) 的有向边。

本题只有一组测试数据。

## 输出格式

输出一行一个整数，表示从 \(1\) 号结点到 \(n\) 号结点的最短路径长度；若不可达，输出 `-1`。

## 样例输入

```text
4 5
1 2 1
1 3 4
2 3 2
2 4 6
3 4 3
```

## 样例输出

```text
6
```

## 数据范围

- \(1 <= n <= 10^5\)，\(0 <= m <= 2 x 10^5\)。
- \(1 <= u,v <= n\)，\(0 <= w <= 10^9\)。
- 允许重边和自环，不保证图连通。
- 最短路径长度可能超过 32 位整数范围，请使用 64 位整数保存；所有输出均为整数。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    long long d;
    int v;
} Item;

static void push(Item *heap, int *size, long long d, int v) {
    int i = ++(*size);
    heap[i].d = d;
    heap[i].v = v;
    while (i > 1) {
        int p = i / 2;
        if (heap[p].d < heap[i].d || (heap[p].d == heap[i].d && heap[p].v <= heap[i].v)) break;
        Item t = heap[p];
        heap[p] = heap[i];
        heap[i] = t;
        i = p;
    }
}

static Item pop(Item *heap, int *size) {
    Item res = heap[1];
    heap[1] = heap[*size];
    (*size)--;
    int i = 1;
    while (1) {
        int l = i * 2, r = l + 1, s = i;
        if (l <= *size && (heap[l].d < heap[s].d || (heap[l].d == heap[s].d && heap[l].v < heap[s].v))) s = l;
        if (r <= *size && (heap[r].d < heap[s].d || (heap[r].d == heap[s].d && heap[r].v < heap[s].v))) s = r;
        if (s == i) break;
        Item t = heap[s];
        heap[s] = heap[i];
        heap[i] = t;
        i = s;
    }
    return res;
}

int main(void) {
    int n, m;
    if (scanf("%d %d", &n, &m) != 2) return 0;

    int *head = (int *)malloc((n + 1) * sizeof(int));
    int *to = (int *)malloc((m + 1) * sizeof(int));
    int *nxt = (int *)malloc((m + 1) * sizeof(int));
    long long *w = (long long *)malloc((m + 1) * sizeof(long long));
    for (int i = 1; i <= n; i++) head[i] = -1;

    for (int i = 0; i < m; i++) {
        int u, v;
        long long wt;
        scanf("%d %d %lld", &u, &v, &wt);
        to[i] = v;
        w[i] = wt;
        nxt[i] = head[u];
        head[u] = i;
    }

    const long long INF = 4000000000000000000LL;
    long long *dist = (long long *)malloc((n + 1) * sizeof(long long));
    for (int i = 1; i <= n; i++) dist[i] = INF;

    Item *heap = (Item *)malloc((m + 2) * sizeof(Item));
    int hsize = 0;
    dist[1] = 0;
    push(heap, &hsize, 0, 1);

    // 邻接表加二叉堆优化的 Dijkstra
    while (hsize > 0) {
        Item cur = pop(heap, &hsize);
        int u = cur.v;
        if (cur.d != dist[u]) continue; // 跳过堆中的过期记录
        for (int e = head[u]; e != -1; e = nxt[e]) {
            int v = to[e];
            long long nd = cur.d + w[e];
            if (nd < dist[v]) {
                dist[v] = nd;
                push(heap, &hsize, nd, v);
            }
        }
    }

    printf("%lld\n", dist[n] == INF ? -1LL : dist[n]);
    free(head);
    free(to);
    free(nxt);
    free(w);
    free(dist);
    free(heap);
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <utility>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<vector<pair<int, long long>>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        long long w;
        cin >> u >> v >> w;
        adj[u].push_back(make_pair(v, w));
    }

    const long long INF = (1LL << 62);
    vector<long long> dist(n + 1, INF);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    dist[1] = 0;
    pq.push(make_pair(0LL, 1));

    // 小根堆优化的 Dijkstra
    while (!pq.empty()) {
        pair<long long, int> cur = pq.top();
        pq.pop();
        long long d = cur.first;
        int u = cur.second;
        if (d != dist[u]) continue; // 忽略已经过期的堆元素
        for (size_t i = 0; i < adj[u].size(); i++) {
            int v = adj[u][i].first;
            long long w = adj[u][i].second;
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push(make_pair(dist[v], v));
            }
        }
    }

    cout << (dist[n] == INF ? -1LL : dist[n]) << '\n';
    return 0;
}
```

### Python

```python
import sys
import heapq


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n + 1)]
    p = 2
    for _ in range(m):
        u, v, w = data[p], data[p + 1], data[p + 2]
        p += 3
        adj[u].append((v, w))

    inf = 10 ** 30
    dist = [inf] * (n + 1)
    dist[1] = 0
    pq = [(0, 1)]
    # 小根堆优化的 Dijkstra
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))

    print(-1 if dist[n] == inf else dist[n])


if __name__ == "__main__":
    main()
```
