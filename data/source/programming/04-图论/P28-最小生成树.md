# P28 最小生成树

- **标签**：#图论 #最小生成树 #并查集
- **难度**：★★★☆☆
- **来源**：经典算法题（Kruskal / Prim）

## 题目描述

给定一张包含 \(n\) 个结点、\(m\) 条边的无向图，结点编号为 \(1\) 到 \(n\)，每条边有一个正整数边权。请计算该图最小生成树的总权值。

最小生成树是原图的一棵生成树，包含图中全部 \(n\) 个结点和恰好 \(n-1\) 条边，并且总边权最小。图中允许重边和自环，自环不会出现在生成树中，可以忽略。若图不连通，则不存在生成树，输出 `-1`。

特别地，当 \(n=1\) 时，最小生成树为空树，总权值为 `0`。

## 输入格式

第一行包含两个整数 \(n\) 和 \(m\)，分别表示结点数和边数。

接下来 \(m\) 行，每行包含三个整数 \(u\)、\(v\)、\(w\)，表示一条连接 \(u\) 和 \(v\)、边权为 \(w\) 的无向边。

本题只有一组测试数据。

## 输出格式

输出一行一个整数，表示最小生成树的总权值；若图不连通，输出 `-1`。

## 样例输入

```text
4 5
1 2 1
1 3 4
2 3 2
2 4 5
3 4 3
```

## 样例输出

```text
6
```

## 数据范围

- \(1 <= n <= 10^5\)，\(1 <= m <= 2 x 10^5\)。
- \(1 <= u,v <= n\)，\(1 <= w <= 10^9\)。
- 允许重边和自环，不保证图连通。
- 最小生成树的总权值可能超过 32 位整数范围，请使用 64 位整数保存；所有输出均为整数。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int u, v;
    long long w;
} Edge;

static int parent[100005], rnk[100005];

static int find(int x) {
    while (parent[x] != x) {
        parent[x] = parent[parent[x]];
        x = parent[x];
    }
    return x;
}

static int cmp(const void *a, const void *b) {
    const Edge *x = (const Edge *)a;
    const Edge *y = (const Edge *)b;
    if (x->w < y->w) return -1;
    if (x->w > y->w) return 1;
    return 0;
}

int main(void) {
    int n, m;
    if (scanf("%d %d", &n, &m) != 2) return 0;

    Edge *edges = (Edge *)malloc((m + 1) * sizeof(Edge));
    for (int i = 0; i < m; i++) {
        scanf("%d %d %lld", &edges[i].u, &edges[i].v, &edges[i].w);
    }
    for (int i = 1; i <= n; i++) {
        parent[i] = i;
        rnk[i] = 0;
    }

    qsort(edges, m, sizeof(Edge), cmp);
    long long total = 0;
    int used = 0;

    // Kruskal：按边权从小到大加入不形成环的边
    for (int i = 0; i < m && used < n - 1; i++) {
        int ru = find(edges[i].u);
        int rv = find(edges[i].v);
        if (ru == rv) continue;
        if (rnk[ru] < rnk[rv]) {
            int t = ru;
            ru = rv;
            rv = t;
        }
        parent[rv] = ru;
        if (rnk[ru] == rnk[rv]) rnk[ru]++;
        total += edges[i].w;
        used++;
    }

    if (used == n - 1) printf("%lld\n", total);
    else puts("-1");
    free(edges);
    return 0;
}
```

### C++

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Edge {
    int u, v;
    long long w;
    bool operator<(const Edge &other) const {
        return w < other.w;
    }
};

struct DSU {
    vector<int> parent, rankValue;

    explicit DSU(int n) : parent(n + 1), rankValue(n + 1, 0) {
        for (int i = 1; i <= n; i++) parent[i] = i;
    }

    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

    bool unite(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;
        if (rankValue[ra] < rankValue[rb]) swap(ra, rb);
        parent[rb] = ra;
        if (rankValue[ra] == rankValue[rb]) rankValue[ra]++;
        return true;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<Edge> edges(m);
    for (int i = 0; i < m; i++) {
        cin >> edges[i].u >> edges[i].v >> edges[i].w;
    }

    sort(edges.begin(), edges.end());
    DSU dsu(n);
    long long total = 0;
    int used = 0;

    // Kruskal：依次尝试加入当前权值最小的边
    for (int i = 0; i < m && used < n - 1; i++) {
        if (dsu.unite(edges[i].u, edges[i].v)) {
            total += edges[i].w;
            used++;
        }
    }

    if (used == n - 1) cout << total << '\n';
    else cout << -1 << '\n';
    return 0;
}
```

### Python

```python
import sys


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    edges = []
    p = 2
    for _ in range(m):
        u, v, w = data[p], data[p + 1], data[p + 2]
        p += 3
        edges.append((w, u, v))
    edges.sort()

    parent = list(range(n + 1))
    rank = [0] * (n + 1)

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    total = 0
    used = 0
    # Kruskal：按边权从小到大加入不形成环的边
    for w, u, v in edges:
        ru, rv = find(u), find(v)
        if ru == rv:
            continue
        if rank[ru] < rank[rv]:
            ru, rv = rv, ru
        parent[rv] = ru
        if rank[ru] == rank[rv]:
            rank[ru] += 1
        total += w
        used += 1
        if used == n - 1:
            break

    print(total if used == n - 1 else -1)


if __name__ == "__main__":
    main()
```
