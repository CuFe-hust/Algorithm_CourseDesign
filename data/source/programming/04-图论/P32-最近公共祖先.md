# P32 最近公共祖先

- **标签**：#图论 #LCA #倍增
- **难度**：★★★★☆
- **来源**：经典算法题（LCA / 倍增）

## 题目描述

给定一棵有 n 个结点的树，结点编号为 1 到 n，并指定其中一个结点 root 作为树根。树是无向连通图，任意两个结点之间有且仅有一条简单路径。

对于两个结点 u 和 v，它们的最近公共祖先（LCA）定义为：同时是 u 的祖先和 v 的祖先的结点中，深度最大的那个结点。这里约定一个结点也是它自己的祖先；根结点的深度为 1，其子结点的深度为 2，依此类推。

现在给出 q 次询问，每次询问两个结点 u 和 v，请回答它们的最近公共祖先的编号。

## 输入格式

第一行包含两个整数 n 和 root，分别表示结点个数和根结点的编号。

接下来 n - 1 行，每行包含两个整数 u 和 v，表示结点 u 与结点 v 之间有一条边。这些边保证构成一棵树。当 n = 1 时，这一部分为空。

接下来一行包含一个整数 q，表示询问次数。

接下来 q 行，每行包含两个整数 u 和 v，表示一次询问。

## 输出格式

输出 q 行，每行一个整数，依次表示每次询问的最近公共祖先的结点编号。

## 样例输入
```text
5 1
1 2
1 3
2 4
2 5
2
4 5
3 4
```
## 样例输出
```text
2
1
```
## 数据范围

- 1 ≤ n ≤ 10^5
- 1 ≤ root ≤ n
- 1 ≤ q ≤ 10^5
- 1 ≤ u, v ≤ n
- 输入保证 n - 1 条边构成一棵以 root 为根的树（连通且无环）
- 询问中 u 与 v 可以相同，此时它们的最近公共祖先就是该结点本身

## 标准实现

### C
```c
#include <stdio.h>

#define MAXN 100005
#define LOG 17

static int head[MAXN], to[2 * MAXN], nxt[2 * MAXN];
static int depth[MAXN], up[LOG][MAXN], que[MAXN];

int main(void) {
    int n, root;
    if (scanf("%d %d", &n, &root) != 2) return 0;
    for (int i = 1; i <= n; i++) head[i] = -1;
    int ecnt = 0;
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        scanf("%d %d", &u, &v);
        to[ecnt] = v; nxt[ecnt] = head[u]; head[u] = ecnt++;
        to[ecnt] = u; nxt[ecnt] = head[v]; head[v] = ecnt++;
    }

    for (int i = 0; i <= n; i++) {
        depth[i] = 0;
        up[0][i] = 0;
    }
    /* BFS 求深度与一级祖先 */
    int front = 0, back = 0;
    que[back++] = root;
    up[0][root] = root;
    while (front < back) {
        int u = que[front++];
        for (int e = head[u]; e != -1; e = nxt[e]) {
            int v = to[e];
            if (v == up[0][u]) continue;
            up[0][v] = u;
            depth[v] = depth[u] + 1;
            que[back++] = v;
        }
    }
    /* up[k][v] 为 v 向上走 2^k 步到达的祖先 */
    for (int k = 1; k < LOG; k++)
        for (int v = 1; v <= n; v++)
            up[k][v] = up[k - 1][up[k - 1][v]];

    int q;
    scanf("%d", &q);
    while (q--) {
        int u, v;
        scanf("%d %d", &u, &v);
        if (depth[u] < depth[v]) {
            int t = u;
            u = v;
            v = t;
        }
        /* 先对齐深度，再同时上跳 */
        int diff = depth[u] - depth[v];
        for (int k = 0; k < LOG; k++)
            if ((diff >> k) & 1) u = up[k][u];
        if (u != v) {
            for (int k = LOG - 1; k >= 0; k--)
                if (up[k][u] != up[k][v]) {
                    u = up[k][u];
                    v = up[k][v];
                }
            u = up[0][u];
        }
        printf("%d\n", u);
    }
    return 0;
}
```

### C++
```cpp
#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, root;
    if (!(cin >> n >> root)) return 0;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int LOG = 1;
    while ((1 << LOG) <= n) LOG++;
    // 倍增预处理：up[k][v] 是 v 的 2^k 级祖先
    vector<vector<int>> up(LOG, vector<int>(n + 1, 0));
    vector<int> depth(n + 1, 0), stk;
    stk.push_back(root);
    up[0][root] = root;
    while (!stk.empty()) {
        int u = stk.back();
        stk.pop_back();
        for (int v : adj[u]) {
            if (v == up[0][u]) continue;
            up[0][v] = u;
            depth[v] = depth[u] + 1;
            stk.push_back(v);
        }
    }
    for (int k = 1; k < LOG; k++)
        for (int v = 1; v <= n; v++)
            up[k][v] = up[k - 1][up[k - 1][v]];

    int q;
    cin >> q;
    while (q--) {
        int u, v;
        cin >> u >> v;
        if (depth[u] < depth[v]) swap(u, v);
        // 先把较深结点提到同一深度
        int diff = depth[u] - depth[v];
        for (int k = 0; k < LOG; k++)
            if ((diff >> k) & 1) u = up[k][u];
        if (u != v) {
            // 一起上跳，直到两者的祖先相同
            for (int k = LOG - 1; k >= 0; k--)
                if (up[k][u] != up[k][v]) {
                    u = up[k][u];
                    v = up[k][v];
                }
            u = up[0][u];
        }
        cout << u << "\n";
    }
    return 0;
}
```

### Python
```python
import sys


def main():
    data = sys.stdin.buffer.read().split()
    pos = 0
    n = int(data[pos]); pos += 1
    root = int(data[pos]); pos += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u = int(data[pos]); v = int(data[pos + 1]); pos += 2
        adj[u].append(v)
        adj[v].append(u)

    LOG = max(1, n.bit_length())
    # 倍增：up[k][v] 是 v 的 2^k 级祖先
    up = [[0] * (n + 1) for _ in range(LOG)]
    depth = [0] * (n + 1)
    up[0][root] = root
    stack = [root]
    while stack:
        u = stack.pop()
        pu = up[0][u]
        du = depth[u] + 1
        for v in adj[u]:
            if v == pu:
                continue
            up[0][v] = u
            depth[v] = du
            stack.append(v)

    for k in range(1, LOG):
        prev = up[k - 1]
        cur = up[k]
        for v in range(1, n + 1):
            cur[v] = prev[prev[v]]

    q = int(data[pos]); pos += 1
    out = []
    for _ in range(q):
        u = int(data[pos]); v = int(data[pos + 1]); pos += 2
        if depth[u] < depth[v]:
            u, v = v, u
        # 先把 u 提到与 v 相同深度
        diff = depth[u] - depth[v]
        k = 0
        while diff:
            if diff & 1:
                u = up[k][u]
            diff >>= 1
            k += 1
        if u != v:
            # 一起上跳，最后再走一步即为 LCA
            for k in range(LOG - 1, -1, -1):
                if up[k][u] != up[k][v]:
                    u = up[k][u]
                    v = up[k][v]
            u = up[0][u]
        out.append(str(u))
    sys.stdout.write("\n".join(out) + "\n")


main()
```
