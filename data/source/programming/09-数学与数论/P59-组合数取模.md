# P59 组合数取模

- **标签**：#数学 #组合数学 #快速幂
- **难度**：★★★★☆
- **来源**：经典算法题（组合数取模）

## 题目描述

给定 T 组询问，每组询问包含两个整数 n 和 m。记 C(n, m) 为从 n 个不同元素中取出 m 个元素的方案数，其定义为

C(n, m) = n! / (m! * (n - m)!)，

其中 0! = 1。本题保证 0 <= m <= n。

请对每组询问计算 C(n, m) mod 1000000007 的值。

## 输入格式

第一行，一个整数 T，表示询问组数。

接下来 T 行，每行两个整数 n 和 m，用一个空格分隔。

## 输出格式

输出 T 行，每行一个整数，依次表示每组询问的答案 C(n, m) mod 1000000007。

## 样例输入

```text
2
5 2
10 3
```

## 样例输出

```text
10
120
```

## 数据范围

- 1 <= T <= 10^4
- 0 <= m <= n <= 10^6
- 模数为 1000000007
- 输入均为整数，没有多余的空行。

## 标准实现

### C

```c
#include <stdio.h>

#define MAXN 1000000
#define MOD 1000000007

static long long fact[MAXN + 1];
static long long inv_fact[MAXN + 1];

long long mod_pow(long long base, long long exp) {
    long long result = 1;
    while (exp > 0) {
        if (exp & 1) result = result * base % MOD;
        base = base * base % MOD;
        exp >>= 1;
    }
    return result;
}

int main(void) {
    int T;
    if (scanf("%d", &T) != 1) return 0;
    static int ns[10000], ms[10000];
    int max_n = 0;
    for (int i = 0; i < T; ++i) {
        scanf("%d %d", &ns[i], &ms[i]);
        if (ns[i] > max_n) max_n = ns[i];
    }
    fact[0] = 1; /* 预处理阶乘与阶乘的逆元 */
    for (int i = 1; i <= max_n; ++i) fact[i] = fact[i - 1] * i % MOD;
    inv_fact[max_n] = mod_pow(fact[max_n], MOD - 2);
    for (int i = max_n; i >= 1; --i) inv_fact[i - 1] = inv_fact[i] * i % MOD;
    for (int i = 0; i < T; ++i) {
        long long ans = fact[ns[i]] * inv_fact[ms[i]] % MOD;
        ans = ans * inv_fact[ns[i] - ms[i]] % MOD;
        printf("%lld\n", ans);
    }
    return 0;
}
```

### Python

```python
import sys

MOD = 1000000007

def mod_pow(base, exp):
    result = 1
    while exp > 0:
        if exp & 1:
            result = result * base % MOD
        base = base * base % MOD
        exp >>= 1
    return result

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    t = data[0]
    queries = []
    max_n = 0
    pos = 1
    for _ in range(t):
        n, m = data[pos], data[pos + 1]
        pos += 2
        queries.append((n, m))
        if n > max_n:
            max_n = n
    fact = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i - 1] * i % MOD
    inv_fact = [1] * (max_n + 1)
    inv_fact[max_n] = mod_pow(fact[max_n], MOD - 2)  # 逆元
    for i in range(max_n, 0, -1):
        inv_fact[i - 1] = inv_fact[i] * i % MOD
    out = []
    for n, m in queries:
        ans = fact[n] * inv_fact[m] % MOD * inv_fact[n - m] % MOD
        out.append(str(ans))
    sys.stdout.write("\n".join(out) + "\n")

if __name__ == "__main__":
    main()
```
