# P03 分数背包

- **标签**：#贪心算法 #背包问题
- **难度**：★★☆☆☆
- **来源**：经典算法题（Fractional Knapsack）

## 题目描述

有 n 件物品和一个容量为 W 的背包。第 i 件物品的重量为 w_i，价值为 v_i。

与 0/1 背包不同，本题中的物品可以被任意分割：若取走某件物品的一部分，获得的价值按取走重量占该物品总重量的比例计算。例如，一件重量为 w、价值为 v 的物品，取走重量 x（0 <= x <= w）的部分，可获得价值 x * v / w。

背包中物品的总重量不能超过 W。求能够装入背包的物品的最大总价值。物品可以只取一部分，也可以不取；背包不要求装满。

## 输入格式

第一行包含两个整数 n 和 W，分别表示物品数量和背包容量。

接下来 n 行，每行包含两个整数 w_i 和 v_i，用空格分隔，表示第 i 件物品的重量和价值。

## 输出格式

输出一行，表示最大总价值，四舍五入保留 2 位小数。

输出必须包含小数点和小数部分，即使小数部分为 0 也要输出两位小数（例如 240.00）。四舍五入规则为：第三位小数大于等于 5 时向第二位小数进位，否则舍去。

## 样例输入

```text
3 50
10 60
20 100
30 120
```

## 样例输出

```text
240.00
```

## 数据范围

- 1 <= n <= 1000
- 1 <= W <= 10^6
- 1 <= w_i <= 10^6
- 0 <= v_i <= 10^6
- 所有输入均为整数，没有多组数据。
- 最大总价值不超过 10^9。

## 标准实现

### C

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    long long w, v;
} Item;

int cmp(const void *a, const void *b) {
    const Item *x = (const Item *)a;
    const Item *y = (const Item *)b;
    long long left = x->v * y->w;
    long long right = y->v * x->w;
    if (left != right) return left > right ? -1 : 1;
    if (x->w != y->w) return x->w < y->w ? -1 : 1;
    return 0;
}

int main(void) {
    int n;
    long long W;
    if (scanf("%d %lld", &n, &W) != 2) return 0;
    Item *a = (Item *)malloc(sizeof(Item) * (size_t)n);
    if (a == NULL) return 0;
    for (int i = 0; i < n; ++i) {
        if (scanf("%lld %lld", &a[i].w, &a[i].v) != 2) {
            free(a);
            return 0;
        }
    }
    qsort(a, (size_t)n, sizeof(Item), cmp);

    long long full = 0, rem = W, num = 0, den = 1;
    int has_frac = 0;
    for (int i = 0; i < n && rem > 0; ++i) {
        if (a[i].w <= rem) {
            rem -= a[i].w;
            full += a[i].v;
        } else {
            num = a[i].v * rem; /* 取走剩余容量对应的部分 */
            den = a[i].w;
            has_frac = 1;
            rem = 0;
        }
    }

    long long cents = full * 100;
    if (has_frac) {
        long long scaled = num * 100;
        cents += scaled / den;
        if ((scaled % den) * 2 >= den) ++cents; /* 四舍五入到分 */
    }
    printf("%lld.%02lld\n", cents / 100, cents % 100);
    free(a);
    return 0;
}
```

### C++

```cpp
#include <algorithm>
#include <iomanip>
#include <iostream>
#include <vector>
using namespace std;

struct Item {
    long long w, v;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long W;
    if (!(cin >> n >> W)) return 0;
    vector<Item> items(n);
    for (auto &item : items) cin >> item.w >> item.v;

    sort(items.begin(), items.end(), [](const Item &a, const Item &b) {
        long long left = a.v * b.w;
        long long right = b.v * a.w;
        if (left != right) return left > right; // 单位价值高者优先
        return a.w < b.w;
    });

    long long full = 0, rem = W, num = 0, den = 1;
    bool hasFrac = false;
    for (const auto &item : items) {
        if (rem == 0) break;
        if (item.w <= rem) {
            rem -= item.w;
            full += item.v;
        } else {
            num = item.v * rem;
            den = item.w;
            hasFrac = true;
            rem = 0;
        }
    }

    long long cents = full * 100;
    if (hasFrac) {
        long long scaled = num * 100;
        cents += scaled / den;
        if ((scaled % den) * 2 >= den) ++cents; // 四舍五入到分
    }
    cout << cents / 100 << '.' << setfill('0') << setw(2) << cents % 100 << '\n';
    return 0;
}
```

### Python

```python
import sys
from functools import cmp_to_key

def cmp(x, y):
    left = x[1] * y[0]
    right = y[1] * x[0]
    if left != right:
        return -1 if left > right else 1
    if x[0] != y[0]:
        return -1 if x[0] < y[0] else 1
    return 0

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    W = int(data[1])
    items = []
    pos = 2
    for _ in range(n):
        w = int(data[pos])
        v = int(data[pos + 1])
        pos += 2
        items.append((w, v))
    items.sort(key=cmp_to_key(cmp))

    full = 0
    rem = W
    num = 0
    den = 1
    has_frac = False
    for w, v in items:
        if rem == 0:
            break
        if w <= rem:
            rem -= w
            full += v
        else:
            num = v * rem
            den = w
            has_frac = True
            rem = 0

    cents = full * 100
    if has_frac:
        scaled = num * 100
        cents += scaled // den
        if (scaled % den) * 2 >= den:
            cents += 1
    print(f"{cents // 100}.{cents % 100:02d}")

if __name__ == "__main__":
    main()
```
